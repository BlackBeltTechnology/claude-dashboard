**Detection Strategy:**

Parallel Task calls appear in ONE assistant message, so SubagentNodes created from that message share the same `parentId`. Use this to detect parallel groups.

Add a helper function at the top of graphLayout.ts (after imports, before `convertSessionToGraph`):

```typescript
/**
 * Detect parallel subagent groups by examining SubagentNode parentIds.
 * Multiple SubagentNodes with the same parentId were invoked in the same
 * assistant message = parallel invocation.
 *
 * Returns: Map of parentId -> array of agentIds (SubagentNode.agentId values)
 * Only includes groups with 2+ parallel subagents.
 */
function detectParallelSubagentGroups(session: Session): Map<string, string[]> {
  const byParent = new Map<string, string[]>();

  for (const node of session.nodes) {
    if (node.type === 'subagent' && node.parentId) {
      const existing = byParent.get(node.parentId);
      if (existing) {
        existing.push(node.agentId);
      } else {
        byParent.set(node.parentId, [node.agentId]);
      }
    }
  }

  // Filter to only groups with 2+ parallel subagents
  const parallelGroups = new Map<string, string[]>();
  for (const [parentId, agentIds] of byParent) {
    if (agentIds.length > 1) {
      parallelGroups.set(parentId, agentIds);
    }
  }

  return parallelGroups;
}
```
